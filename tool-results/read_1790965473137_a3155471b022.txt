   210→// ─── خواندن خودکار پیامک‌های بانکی (فقط اندروید) ───
   211→function SmsAutoSync() {
   212→  const { toast } = useToast();
   213→  const qc = useQueryClient();
   214→  const [mounted, setMounted] = useState(false);
   215→  const [perm, setPerm] = useState<SmsPermState | null>(null);
   216→  const [syncing, setSyncing] = useState(false);
   217→  const [lastSync, setLastSync] = useState(0);
   218→  const [autoImport, setAutoImport] = useState(true);
   219→  const [convertRial, setConvertRial] = useState(true);
   220→
   221→  const syncingRef = useRef(false);
   222→  const optsRef = useRef({ autoImport: true, convertRial: true });
   223→  useEffect(() => {
   224→    optsRef.current = { autoImport, convertRial };
   225→  }, [autoImport, convertRial]);
   226→
   227→  const runSync = useCallback(
   228→    async (silent: boolean) => {
   229→      if (syncingRef.current) return;
   230→      syncingRef.current = true;
   231→      setSyncing(true);
   232→      try {
   233→        const res = await syncBankSms({ ...optsRef.current });
   234→        setLastSync(await getLastSmsSync());
   235→        qc.invalidateQueries({ queryKey: ["sms-logs"] });
   236→        qc.invalidateQueries({ queryKey: ["transactions"] });
   237→        qc.invalidateQueries({ queryKey: ["stats"] });
   238→        if (res.ok) {
   239→          const parts: string[] = [];
   240→          if (res.imported > 0) parts.push(`${toFa(res.imported)} تراکنش خودکار ثبت شد`);
   241→          if (res.queued > 0) parts.push(`${toFa(res.queued)} پیامک در صف بررسی است`);
   242→          const desc = parts.length > 0 ? parts.join(" — ") : "پیامک بانکی جدیدی نبود";
   243→          if (!silent || res.imported > 0 || res.queued > 0) {
   244→            toast({ title: "پیامک‌ها بررسی شد", description: desc });
   245→          }
   246→        } else if (!silent && res.reason === "permission") {
   247→          toast({ title: "دسترسی پیامک داده نشده است", variant: "destructive" });
   248→        } else if (!silent && res.reason === "error") {
   249→          toast({ title: "خطا در خواندن پیامک‌ها", variant: "destructive" });
   250→        }
   251→      } catch {
   252→        if (!silent) toast({ title: "خطا در همگام‌سازی", variant: "destructive" });
   253→      } finally {
   254→        syncingRef.current = false;
   255→        setSyncing(false);
   256→      }
   257→    },
   258→    [qc, toast]
   259→  );
   260→
   261→  // بارگذاری اولیه + sync خودکار در باز شدن تب
   262→  useEffect(() => {
   263→    if (!isNativeAndroid()) return;
   264→    setMounted(true);
   265→    void (async () => {
   266→      const [p, ls, ai, cr] = await Promise.all([
   267→        getSmsPermission(),
   268→        getLastSmsSync(),
   269→        getMeta<boolean>("smsAutoImport"),
   270→        getMeta<boolean>("smsConvertRial"),
   271→      ]);
   272→      setPerm(p);
   273→      setLastSync(ls || 0);
   274→      if (ai === false) setAutoImport(false);
   275→      if (cr === false) setConvertRial(false);
   276→      if (p === "granted") void runSync(true);
   277→    })();
   278→  }, [runSync]);
   279→
   280→  // گوش دادن به پیامک زنده (وقتی برنامه باز است)
   281→  useEffect(() => {
   282→    if (!isNativeAndroid()) return;
   283→    let handle: PluginListenerHandle | null = null;
   284→    let timer: ReturnType<typeof setTimeout> | null = null;
   285→    HesabSms.addListener("smsReceived", () => {
   286→      if (timer) clearTimeout(timer);
   287→      timer = setTimeout(() => void runSync(true), 3000);
   288→    })
   289→      .then((h) => {
   290→        handle = h;
   291→      })
   292→      .catch(() => {});
   293→    return () => {
   294→      if (timer) clearTimeout(timer);
   295→      void handle?.remove();
   296→    };
   297→  }, [runSync]);
   298→
   299→  const askPermission = async () => {
   300→    const p = await requestSmsPermission();
   301→    setPerm(p);
   302→    if (p === "granted") {
   303→      toast({ title: "دسترسی پیامک فعال شد" });
   304→      void runSync(false);
   305→    } else if (p === "denied") {
   306→      toast({
   307→        title: "دسترسی رد شد",
   308→        description: "از تنظیمات گوشی ← برنامه‌ها ← حساب‌یار ← مجوزها، دسترسی پیامک را فعال کنید",
   309→        variant: "destructive",
   310→      });
   311→    }
   312→  };
   313→
   314→  if (!mounted || perm === null) return null;
   315→
   316→  return (
   317→    <Card>
   318→      <CardHeader className="pb-2">
   319→        <CardTitle className="flex items-center gap-2 text-base">
   320→          <BellRing className="h-4 w-4 text-emerald-700" />
   321→          خواندن خودکار پیامک بانکی
   322→        </CardTitle>
   323→      </CardHeader>
   324→      <CardContent className="space-y-3">
   325→        {perm !== "granted" ? (
   326→          <div className="flex items-start gap-3 rounded-xl border bg-[rgba(254,243,199,0.45)] p-3">
   327→            <ShieldAlert className="mt-0.5 h-5 w-5 shrink-0 text-amber-600" />
   328→            <div className="space-y-2">
   329→              <p className="text-xs leading-5">
   330→                برای ثبت خودکار واریز و برداشت، برنامه باید پیامک‌های بانکی را بخواند.
   331→                هیچ داده‌ای از گوشی شما خارج نمی‌شود و همه‌چیز روی خود دستگاه ذخیره می‌شود.
   332→              </p>
   333→              <Button size="sm" onClick={askPermission} className="bg-emerald-700 hover:bg-emerald-800">
   334→                <ShieldCheck className="ml-1 h-4 w-4" /> اجازه دسترسی به پیامک‌ها
   335→              </Button>
   336→            </div>
   337→          </div>
   338→        ) : (
   339→          <>
   340→            <div className="flex items-center justify-between gap-2">
   341→              <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   342→                <ShieldCheck className="h-4 w-4 text-emerald-600" />
   343→                <span>دسترسی فعال — {lastSync > 0 ? `آخرین بررسی: ${new Date(lastSync).toLocaleString("fa-IR")}` : "هنوز بررسی نشده"}</span>
   344→              </div>
   345→              <Button size="sm" variant="outline" className="h-7 text-xs" disabled={syncing} onClick={() => void runSync(false)}>
   346→                <RefreshCw className={`ml-1 h-3 w-3 ${syncing ? "animate-spin" : ""}`} />
   347→                {syncing ? "در حال بررسی..." : "بررسی الان"}
   348→              </Button>
   349→            </div>
   350→            <div className="flex items-center justify-between rounded-xl border p-3">
   351→              <div className="pl-3">
   352→                <div className="text-sm font-medium">ثبت خودکار تراکنش‌ها</div>
   353→                <p className="text-[11px] leading-4 text-muted-foreground">
   354→                  پیامک‌های واریز/برداشت با مبلغ واضح، بدون دست وارد شوند؛ موارد نامطمئن در صف بررسی می‌مانند
   355→                </p>
   356→              </div>
   357→              <Switch
   358→                checked={autoImport}
   359→                onCheckedChange={(v) => {
   360→                  setAutoImport(v);
   361→                  void setMeta("smsAutoImport", v);
   362→                }}
   363→              />
   364→            </div>
   365→            <div className="flex items-center justify-between rounded-xl border p-3">
   366→              <div className="pl-3">
   367→                <div className="text-sm font-medium">تبدیل ریال به تومان</div>
   368→                <p className="text-[11px] leading-4 text-muted-foreground">
   369→                  مبالغ پیامک بانک‌ها معمولاً ریال است؛ هنگام ثبت خودکار تقسیم بر ۱۰ می‌شود
   370→                </p>
   371→              </div>
   372→              <Switch
   373→                checked={convertRial}
   374→                onCheckedChange={(v) => {
   375→                  setConvertRial(v);
   376→                  void setMeta("smsConvertRial", v);
   377→                }}
   378→              />
   379→            </div>
   380→          </>
   381→        )}
   382→      </CardContent>
   383→    </Card>
   384→  );
   385→}
